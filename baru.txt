    function bukaModal(key, type) {
        const data = (type === 'trip') ? tripData[key] : experienceData[key];
        if (!data) return;

        const leftSidebar = document.getElementById('experienceLeftSidebar');
        const mainContent = document.getElementById('modalMainContent');
        const badgeElement = document.getElementById('modalBadge');
        const headerLabel = document.getElementById('itineraryHeader');

        document.getElementById('modalTripTitle').innerText = data.title;
        document.getElementById('modalTripDesc').innerText = data.desc;
        document.getElementById('timelineContent').innerHTML = data.itinerary;

        if (type === 'experience') {
            leftSidebar.classList.remove('hidden');
            mainContent.classList.remove('md:col-span-12');
            mainContent.classList.add('md:col-span-8');
            
            badgeElement.innerText = data.badge || "Experience";
            badgeElement.classList.remove('hidden');
            headerLabel.innerText = "What we'll do :";

            document.getElementById('expDuration').innerText = data.duration;
            document.getElementById('expLanguage').innerText = data.language;
            document.getElementById('expSnack').innerText = data.snack;
            document.getElementById('expBookingRule').innerText = data.bookingRule;
            
            document.getElementById('priceIncStrike').innerText = data.priceIncStrike;
            document.getElementById('priceIncReal').innerText = data.priceIncReal;
            document.getElementById('priceExcStrike').innerText = data.priceExcStrike;
            document.getElementById('priceExcReal').innerText = data.priceExcReal;
        } else {
            leftSidebar.classList.add('hidden');
            mainContent.classList.remove('md:col-span-8');
            mainContent.classList.add('md:col-span-12');
            
            badgeElement.classList.add('hidden');
            headerLabel.innerText = "Rencana Perjalanan / Itinerary";
        }

        let includeHtml = "";
        if (data.include) {
            data.include.forEach(item => { includeHtml += `<li>${item}</li>`; });
        }
        document.getElementById('modalInclude').innerHTML = includeHtml;

        let excludeHtml = "";
        if (data.exclude) {
            data.exclude.forEach(item => { excludeHtml += `<li>${item}</li>`; });
        }
        document.getElementById('modalExclude').innerHTML = excludeHtml;

        const waBaseUrl = "https://wa.me/62895369809676";
        let textMessage = (type === 'experience') 
            ? `Halo Dawn'scape, saya tertarik booking Experience: *${data.title}*.`
            : `Halo Dawn'scape, saya tertarik booking Open Trip: *${data.title}*.`;
        
        document.getElementById('modalBookingBtn').href = `${waBaseUrl}?text=${encodeURIComponent(textMessage)}`;

        const overlay = document.getElementById('itineraryModal');
        const box = document.getElementById('modalBox');
        overlay.classList.remove('hidden');
        setTimeout(() => {
            box.classList.remove('scale-95', 'opacity-0');
            box.classList.add('scale-100', 'opacity-100');
        }, 20);
    }

    function tutupModal() {
        const overlay = document.getElementById('itineraryModal');
        const box = document.getElementById('modalBox');
        box.classList.remove('scale-100', 'opacity-100');
        box.classList.add('scale-95', 'opacity-0');
        setTimeout(() => { overlay.classList.add('hidden'); }, 300);
    }

    window.onclick = function(event) {
        const overlay = document.getElementById('itineraryModal');
        if (event.target === overlay) { tutupModal(); }
    }
</script>
