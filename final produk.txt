        // Fungsi Membuka Modal
function bukaModal(id, type) {
            const data = databaseProduk[id];
            if (!data) return;

            document.getElementById('modalTitle').innerText = data.title;
            document.getElementById('modalPrice').innerText = `${data.price}/pax`;

            const badge = document.getElementById('modalTypeBadge');
            const mpContainer = document.getElementById('modalMeetingPointContainer');
            const mpText = document.getElementById('modalMeetingPointText');

            // Cek tipe dan atur tampilan khusus Open Trip
            if (type === 'experience') {
                badge.innerText = 'Experience';
                badge.className = 'inline-flex items-center px-2.5 py-0.5 rounded-full text-xs font-bold bg-blue-100 text-blue-800 mb-1';
                mpContainer.classList.add('hidden'); // Sembunyikan meeting point untuk experience
            } else {
                badge.innerText = 'Open Trip';
                badge.className = 'inline-flex items-center px-2.5 py-0.5 rounded-full text-xs font-bold bg-emerald-100 text-emerald-800 mb-1';
                
                // Cek jika produk bertipe trip dan memiliki properti data meetingPoint
                if (type === 'trip' && data.meetingPoint) {
                    mpText.innerText = data.meetingPoint;
                    mpContainer.classList.remove('hidden'); // Tampilkan tepat di atas itinerary
                } else {
                    mpContainer.classList.add('hidden');
                }
            }

            // Susun Itinerary HTML
            let itineraryHtml = '';
            data.itinerary.forEach(item => {
                itineraryHtml += `
                    <div class="relative pl-6 group">
                        <div class="absolute -left-[7px] top-1.5 w-3 h-3 rounded-full bg-blue-500 border-2 border-white shadow-sm transition-transform group-hover:scale-125 duration-200"></div>
                        <span class="block text-xs font-bold text-blue-600 mb-0.5">${item.time}</span>
                        <p class="text-sm font-normal text-slate-700 leading-relaxed">${item.activity}</p>
                    </div>
                `;
            });
            document.getElementById('modalItinerary').innerHTML = itineraryHtml;

            // Susun Include HTML
            let includeHtml = '';
            data.include.forEach(item => {
                includeHtml += `<li class="flex items-start"><span class="text-emerald-500 mr-2 font-bold">✓</span>${item}</li>`;
            });
            document.getElementById('modalInclude').innerHTML = includeHtml;

            // Susun Exclude HTML
            let excludeHtml = '';
            data.exclude.forEach(item => {
                excludeHtml += `<li class="flex items-start"><span class="text-rose-400 mr-2 font-bold">✕</span>${item}</li>`;
            });
            document.getElementById('modalExclude').innerHTML = excludeHtml;

            // Atur pesan WhatsApp dinamis
            const waBaseUrl = "https://wa.me/62895369809676";
            let textMessage = (type === 'experience') 
                ? `Halo Dawn'scape, saya tertarik booking Experience: *${data.title}*.`
                : `Halo Dawn'scape, saya tertarik booking Open Trip: *${data.title}*.`;
            
            document.getElementById('modalBookingBtn').href = `${waBaseUrl}?text=${encodeURIComponent(textMessage)}`;

            // Tampilkan Modal Animation overlay
            const overlay = document.getElementById('itineraryModal');
            const box = document.getElementById('modalBox');
            overlay.classList.remove('hidden');
            setTimeout(() => {
                box.classList.remove('scale-95', 'opacity-0');
                box.classList.add('scale-100', 'opacity-100');
            }, 20);
        }

        // Fungsi Menutup Modal
        function tutupModal() {
            const overlay = document.getElementById('itineraryModal');
            const box = document.getElementById('modalBox');
            box.classList.remove('scale-100', 'opacity-100');
            box.classList.add('scale-95', 'opacity-0');
            setTimeout(() => { overlay.classList.add('hidden'); }, 300);
        }

        }
